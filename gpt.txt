"""
opd_pipeline.py (fixed)

Pipeline for:
 - downloading Open Bandit Dataset (OBP)
 - constructing per-item empirical traces
 - estimating priors via StickinessHelper
 - building a simulation environment
 - running contextual Thompson experiments comparing:
     * Progressive (ProgressiveBelief + BayesianLinearModel, pb_weight>0)
     * ContextOnly (same beliefs but pb_weight=0.0)
     * Random baseline (no learning, purely random)
 - logging behavior, computing regret (context-aware oracle), saving agents/results, and running OPE.

Key fixes compared to prior version:
 - Trace generator now mirrors run_contextual_eval.py: base * scale(context) * day_decay
   and sampling via Bernoulli per-day. This ensures the context signal that ProgressiveBelief
   is meant to capture is actually present.
 - Deterministic, reproducible seeding (MASTER_SEED -> per-policy/per-trial/per-arm seeds).
 - Use the same belief initialization across Progressive/ContextOnly; only pb_weight differs.
 - Context-aware oracle (expected_reward_for_arm) used for regret calculation.
 - Deterministic propensity MC (estimate_thompson_probas accepts rng_local).
 - Random baseline implemented as simple RandomAgent (non-learning).
"""

def ensure_obd_downloaded(behavior_policy: str, campaign: str, data_path: Path):
    print(f"Downloading / loading Open Bandit Dataset (policy={behavior_policy}, campaign={campaign}) ...")
    dataset = OpenBanditDataset(behavior_policy=behavior_policy, campaign=campaign, data_path=data_path)
    bandit_feedback = dataset.obtain_batch_bandit_feedback()
    raw_df = dataset.data
    print("Loaded dataset; n_rounds:", bandit_feedback.get("n_rounds", None))
    return dataset, bandit_feedback, raw_df


def to_day_bucket(ts_series):
    try:
        ts = pd.to_datetime(ts_series, unit="s")
    except Exception:
        ts = pd.to_datetime(ts_series)
    days = (ts - ts.min()).dt.days
    return days.values


def _safe_align_vec(vec, target_len):
    v = np.asarray(vec).reshape(-1)
    if v.size == target_len:
        return v.astype(float)
    if v.size < target_len:
        out = np.zeros(target_len, dtype=float)
        out[: v.size] = v
        return out
    return v[:target_len].astype(float)


def _safe_posterior_mean_of_model(model, target_len):
    try:
        pm = model.posterior_mean()
        arr = np.asarray(pm).reshape(-1)
    except Exception:
        return np.zeros(target_len, dtype=float)
    if arr.ndim == 0:
        arr = arr.reshape(1)
    return _safe_align_vec(arr, target_len)

# Context sampling helper (deterministic per-trial RNG)

def sample_context(rng_local):
    if use_synthetic_contexts:
        ix = int(rng_local.integers(0, base_ctx.shape[0]))
        raw = base_ctx[ix]
    else:
        ix = int(rng_local.integers(0, contexts.shape[0]))
        raw = contexts[ix]
    return np.concatenate([[1.0], np.asarray(raw).reshape(-1)])  # length CONTEXT_DIM+1

# -------------------------------
# Generator that mirrors main.py (context-dependent full-trace sampling)
# -------------------------------
def generate_trace_for_arm(arm, context, n_samples=1, rng_local=None):
    """
    Generate n_samples full traces for `arm` where per-day prob = base * scale(context) * day_decay.
    """
    base = empirical_means[arm]  # shape (W,)
    x = np.asarray(context).reshape(-1)
    x = _safe_align_vec(x, CONTEXT_DIM + 1)
    mu = float(true_theta[arm] @ x)
    scale = 0.5 * (1.0 / (1.0 + np.exp(-mu))) + 0.5
    probs = np.clip(base * scale * day_decay, 1e-6, 1 - 1e-6)
    if rng_local is None:
        rng_local = np.random.default_rng(int(MASTER_SEED + 5000))
    samples = rng_local.binomial(1, probs, size=(n_samples, probs.size)).astype(float)
    return samples

# -------------------------------
# Deterministic Thompson-propensity estimator
# -------------------------------
def estimate_thompson_probas(agent, context, B=PROP_MC, rng_local=None, agent_seed: int = None):
    if rng_local is None:
        if agent_seed is None:
            rng_local = np.random.default_rng(MASTER_SEED + 4242)
        else:
            rng_local = np.random.default_rng(int(agent_seed + 4242))
    arms = agent.arms
    K = len(arms)
    x = np.asarray(context).reshape(-1)
    dim_x = x.size
    values = np.full((K, B), -np.inf, dtype=float)
    for i, a in enumerate(arms):
        try:
            model = agent.models[a]
            Lambda = np.asarray(model.Lambda)
            try:
                cov = np.linalg.inv(Lambda)
            except Exception:
                cov = np.linalg.pinv(Lambda + np.eye(Lambda.shape[0]) * 1e-9)
            mean = _safe_posterior_mean_of_model(model, Lambda.shape[0])
        except Exception:
            mean = np.zeros(dim_x, dtype=float)
            cov = np.eye(dim_x, dtype=float) * 1e-6
        if cov.shape[0] != dim_x or cov.shape[1] != dim_x:
            if cov.shape[0] >= dim_x and cov.shape[1] >= dim_x:
                cov = cov[:dim_x, :dim_x]
            else:
                newcov = np.eye(dim_x, dtype=float) * 1e-6
                newcov[: cov.shape[0], : cov.shape[1]] = cov
                cov = newcov
            mean = _safe_align_vec(mean, dim_x)
        else:
            mean = _safe_align_vec(mean, dim_x)
        try:
            bel = agent.beliefs[a]
            try:
                if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(agent, "t", None):
                    bel.compute_posterior(getattr(agent, "t", 0))
            except Exception:
                pass
            pb_mean = float(getattr(bel, "mean", 0.0))
            pb_var = float(getattr(bel, "var", 0.0))
            pb_std = float(np.sqrt(max(pb_var, 1e-12)))
        except Exception:
            pb_mean = 0.0
            pb_std = 1e-6
        try:
            thetas = rng_local.multivariate_normal(mean, cov, size=B)
            thetas = np.atleast_2d(thetas)
            if thetas.shape[1] != dim_x:
                tmp = np.zeros((thetas.shape[0], dim_x), dtype=float)
                tmp[:, : thetas.shape[1]] = thetas[:, : min(thetas.shape[1], dim_x)]
                thetas = tmp
        except Exception:
            thetas = np.tile(mean.reshape(1, -1), (B, 1))
        try:
            preds = thetas @ x
        except Exception:
            preds = np.zeros(B, dtype=float)
        pb_samples = pb_mean + pb_std * rng_local.standard_normal(size=B)
        pw = float(getattr(agent, "pb_weight", PB_WEIGHT_DEFAULT))
        values[i, :] = preds + pw * pb_samples
    chosen = np.argmax(values, axis=0)
    counts = np.bincount(chosen, minlength=K)
    probas = counts.astype(float) / float(B)
    return {arms[i]: float(probas[i]) for i in range(K)}

# -------------------------------
# Simple RandomAgent baseline (non-learning)
# -------------------------------
class RandomAgent:
    def __init__(self, arms, seed: int = 0):
        self.arms = list(arms)
        self.rng = np.random.default_rng(int(seed))
    def act(self, context, n_actions=1):
        k = len(self.arms)
        n = int(n_actions)
        if n >= k:
            return list(self.rng.choice(self.arms, size=k, replace=False))
        return list(self.rng.choice(self.arms, size=n, replace=False))
    def update(self, *args, **kwargs):
        return
    def step(self):
        return

# -------------------------------
# Oracle expected reward consistent with generator
# -------------------------------
def expected_reward_for_arm(arm, context):
    base = empirical_means[arm]
    x = np.asarray(context).reshape(-1)
    x = _safe_align_vec(x, CONTEXT_DIM + 1)
    mu = float(true_theta[arm] @ x)
    scale = 0.5 * (1.0 / (1.0 + np.exp(-mu))) + 0.5
    probs = np.clip(base * scale * day_decay, 1e-6, 1 - 1e-6)
    return 1.0 + probs.sum()

# -------------------------------
# Thompson training + logging (deterministic seeds)
# -------------------------------
def bandit_trial_contextual_with_logging(env, beliefs, horizon, n_actions, policy_name, agent_seed, pb_weight_setting, propensity_mc=PROP_MC):
    """
    policy_name: "Progressive", "ContextOnly", or "Random"
    agent_seed: deterministic int seed for trial
    """
    if policy_name == "Random":
        agent = RandomAgent(env.dists.keys(), seed=agent_seed)
        # For logging compatibility attach placeholders
        agent.models = {a: None for a in env.dists.keys()}
        agent.beliefs = {a: None for a in env.dists.keys()}
        agent.pb_weight = 0.0
        agent.t = 0
    else:
        # Use ProgressiveBelief for both policies; pb_weight toggles progressive contribution
        agent = ContextualBayesianBandit(
            beliefs, dim=CONTEXT_DIM + 1, alpha=1.0, sigma2=0.25, pb_weight=float(pb_weight_setting), seed=agent_seed
        )

    regrets = []
    entropies = []
    logged = []

    rng_prop_base = np.random.default_rng(int(agent_seed + 11111))  # base to derive MC RNGs
    rng_gen_base = np.random.default_rng(int(agent_seed + 22222))   # base for generator RNGs
    rng_ctx_base = np.random.default_rng(int(agent_seed + 33333))   # base for context sampling

    for t in range(horizon):
        rng_t = np.random.default_rng(int(agent_seed + t + 40000))
        context = sample_context(rng_ctx_base) if use_synthetic_contexts else sample_context(rng_ctx_base)

        # sample actions
        actions = agent.act(context, n_actions=n_actions)

        # deterministic MC RNG for propensity estimation (seed depends on agent_seed + t)
        rng_prop = np.random.default_rng(int(agent_seed + t + 50000))
        try:
            if policy_name == "Random":
                probas = {a: 1.0 / len(agent.arms) for a in agent.arms}
            else:
                probas = estimate_thompson_probas(agent, context, B=propensity_mc, rng_local=rng_prop, agent_seed=agent_seed)
        except Exception:
            probas = {a: 1.0 / len(agent.arms) for a in agent.arms}

        for action in actions:
            p_b = max(probas.get(action, 0.0), 1e-12)
            # deterministic generator RNG for this (trial, t, arm)
            arm_idx = selected_items.index(action)
            rng_for_gen = np.random.default_rng(int(agent_seed + t + 60000 + arm_idx))
            # generate trace using same generator as main.py
            trace = generate_trace_for_arm(action, context, n_samples=1, rng_local=rng_for_gen)
            r = float(1.0 + trace.sum())
            logged.append(LoggedStep(x=context.copy(), a=action, r=r, p_b=p_b, t=t))
            # update agent only if it supports learning and is not Random
            if policy_name != "Random":
                try:
                    agent.update(action, trace, context)
                except Exception:
                    pass

        if policy_name != "Random":
            try:
                agent.step()
            except Exception:
                pass

        # compute context-aware oracle regret
        true_rewards = np.array([expected_reward_for_arm(a, context) for a in actions])
        r_opt = max(expected_reward_for_arm(it, context) for it in selected_items)
        regrets.append(np.mean(r_opt - true_rewards))

        _, counts = np.unique(actions, return_counts=True)
        probs = counts / counts.sum()
        entropies.append(entropy(probs))

    return np.array(regrets), np.array(entropies), agent, logged

        except Exception as e:
            print(f"WARNING: failed to save trained agent to {fname}: {e}")

        # store results
        results_here = {"regrets": reg[n][name], "ent": ent[n][name], "trained_agent": last_trained_agent, "logged": last_logged_data}
        # compute OPE (Direct Method) on last logged dataset
        if last_logged_data is None or len(last_logged_data) == 0:
            print(f"{name} | no logged data collected; skipping OPE")
        else:
            arms = list(env.dists.keys())
            dim = CONTEXT_DIM + 1
            dm_model = DirectMethodModel(arms=arms, dim=dim, alpha=1.0, sigma2=0.25)
            dm_model.fit(last_logged_data)

            # evaluation function (greedy deterministic)
            def make_eval_fn(agent):
                def eval_action_fn(x):
                    x = np.asarray(x).reshape(-1)
                    best_val = -np.inf
                    best_arm = None
                    for a in arms:
                        try:
                            if hasattr(agent, "models") and agent.models.get(a, None) is not None:
                                theta = _safe_posterior_mean_of_model(agent.models[a], x.size)
                            else:
                                theta = np.zeros(x.size, dtype=float)
                        except Exception:
                            theta = np.zeros(x.size, dtype=float)
                        if theta.size != x.size:
                            tmp = np.zeros_like(x)
                            tmp[: min(theta.size, x.size)] = theta[: min(theta.size, x.size)]
                            pred = float(tmp @ x)
                        else:
                            pred = float(theta @ x)
                        pb_mean = 0.0
                        try:
                            bel = getattr(agent, "beliefs", {}).get(a, None)
                            if bel is not None:
                                try:
                                    if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(agent, "t", None):
                                        bel.compute_posterior(getattr(agent, "t", 0))
                                except Exception:
                                    pass
                                pb_mean = float(getattr(bel, "mean", 0.0))
                        except Exception:
                            pb_mean = 0.0
                        val = pred + float(getattr(agent, "pb_weight", 0.0)) * pb_mean
                        if val > best_val:
                            best_val = val
                            best_arm = a
                    return best_arm
                return eval_action_fn

            eval_fn = make_eval_fn(last_trained_agent)
            metrics = evaluate_offline(dataset=last_logged_data, arms=arms, eval_action_fn=eval_fn, dm_model=dm_model, clip=10.0, snips=True, n_boot=500)

            print("\n=== OPE for", name, "===")
            for k_metric, v_metric in metrics.items():
                if k_metric != "diagnostics":
                    print(f"{name} | {k_metric}: {v_metric['value']:.6f} (boot mean {v_metric['boot_mean']:.6f} CI95 {v_metric['ci95'][0]:.6f} .. {v_metric['ci95'][1]:.6f})")
            diag = metrics.get("diagnostics", {})
            print(f"{name} | diagnostics: coverage={diag.get('coverage', np.nan):.3f}, ESS={diag.get('ess', np.nan):.1f}, avg_weight={diag.get('avg_weight', np.nan):.3f}")

        # dump results summary per policy
        out_fname = OUTPUT_DIR / f"experiment_results_{name}.pkl"
