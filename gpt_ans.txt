# Fallback simple LoggedStep if package doesn't export it exactly
@dataclass
class SimpleLoggedStep:
    x: np.ndarray
    a: Any
    r: float
    p_b: float
    t: int


# Utility: safe vector alignment (pad / truncate)
def _safe_align_vec(vec: np.ndarray, target_len: int) -> np.ndarray:
    v = np.asarray(vec).reshape(-1)
    if v.size == target_len:
        return v.astype(float)
    if v.size < target_len:
        out = np.zeros(target_len, dtype=float)
        out[: v.size] = v
        return out
    return v[:target_len].astype(float)


def _safe_posterior_mean_of_model(model, target_len: int) -> np.ndarray:
    try:
        pm = model.posterior_mean()
        arr = np.asarray(pm).reshape(-1)
    except Exception:
        return np.zeros(target_len, dtype=float)
    return _safe_align_vec(arr, target_len)


class ModelService:
    """
    Wraps agent loading/saving, prediction, batch updates, environment construction,
    and offline evaluation helpers.
    """

    def __init__(
        self,
        data_dir: str = "data",
        train_pkl: str = "data/synthetic-data-train.pkl",
        eval_pkl: str = "data/synthetic-data-eval.pkl",
        w: int = 59,
        context_dim: int = 4,
        pb_weight: float = 0.6,
        sigma2: float = 0.25,
        rng_seed: int = 1,
    ):
        self.data_dir = data_dir
        self.train_pkl = train_pkl
        self.eval_pkl = eval_pkl
        self.w = w
        self.context_dim = context_dim
        self.pb_weight = pb_weight
        self.sigma2 = sigma2
        self.rng = np.random.default_rng(rng_seed)

        # placeholders
        self.helper: Optional[StickinessHelper] = None
        self.env: Optional[Environment] = None
        self.item_traces: Optional[Dict[str, np.ndarray]] = None
        self.empirical_means: Optional[Dict[str, np.ndarray]] = None
        self.day_decay: Optional[np.ndarray] = None
        self.true_theta: Optional[Dict[str, np.ndarray]] = None

        self.agent: Optional[ContextualBayesianBandit] = None
        self.logged: List[Any] = []  # list of LoggedStep-like objects (for OPE)
        self._load_env_from_eval_pkl_if_exists()

    def _load_pickle(self, path):
        with open(path, "rb") as fh:
            return pickle.load(fh)

    def _load_env_from_eval_pkl_if_exists(self):
        """If eval pkl exists, build EmpiricalDistribution-based env (like your offline experiments)."""
        if os.path.exists(self.eval_pkl):
            data_eval = self._load_pickle(self.eval_pkl)
            # only keep keys that have at least one trace and proper shape
            item_traces = {k: v.astype(float) for k, v in data_eval.items() if v is not None and len(v) > 0}
            if len(item_traces) == 0:
                return
            self.item_traces = item_traces
            # Empirical means
            self.empirical_means = {k: np.mean(v, axis=0) for k, v in item_traces.items()}
            self.day_decay = np.linspace(1.0, 0.6, self.w)
            # build env
            dists = {}
            for uri, traces in item_traces.items():
                seed_int = int(self.rng.integers(0, 2 ** 31 - 1))
                dists[uri] = EmpiricalDistribution(traces.copy(), seed=seed_int)
            self.env = Environment(dists)
            # recreate a deterministic true_theta mapping (useful for simulation/diagnostics)
            rng2 = np.random.default_rng(12345)
            self.true_theta = {uri: rng2.normal(scale=0.8, size=(self.context_dim + 1,)) for uri in item_traces.keys()}
            # If training pkl exists, compute helper for priors
            if os.path.exists(self.train_pkl):
                train_data = self._load_pickle(self.train_pkl)
                train_data = {k: v.astype(float) for k, v in train_data.items() if v is not None and len(v) > 0}
                if len(train_data) > 0:
                    self.helper = StickinessHelper.from_data(train_data)
            return

    # ---------- Agent lifecycle ----------
    def load_agent(self, path: str):
        """Load a pickled trained agent."""
        with open(path, "rb") as fh:
            agent = pickle.load(fh)
        self.agent = agent
        return agent

    def save_agent(self, path: str):
        if self.agent is None:
            raise RuntimeError("No agent loaded to save.")
        with open(path, "wb") as fh:
            pickle.dump(self.agent, fh, protocol=pickle.HIGHEST_PROTOCOL)

    def create_fresh_agent(self, belief_cls=ProgressiveBelief):
        """Initialize an untrained agent using priors from helper and default BayesianLinearModel per arm."""
        if self.env is None:
            raise RuntimeError("Environment not constructed. Need eval pkl in place.")
        if self.helper is None:
            raise RuntimeError("Stickiness helper not available. Provide training pkl or compute priors.")
        beliefs = {}
        for uri in self.env.dists.keys():
            beliefs[uri] = belief_cls(
                prior_mvec=self.helper.prior_mvec.copy(),
                prior_cmat=self.helper.prior_cmat.copy(),
                noise_cmat=self.helper.noise_cmat.copy(),
                cov_estimator="fixed",
                seed=int(self.rng.integers(0, 2 ** 31 - 1)),
            )
        agent = ContextualBayesianBandit(beliefs, dim=self.context_dim + 1, alpha=1.0, sigma2=self.sigma2, pb_weight=self.pb_weight, seed=int(self.rng.integers(0, 2 ** 31 - 1)))
        self.agent = agent
        return agent

    # ---------- Propensity MC for Thompson ----------
    def estimate_thompson_probas(self, context: np.ndarray, B: int = 500) -> Dict[str, float]:
        """Monte-Carlo estimate p_b(a|x) for the currently loaded agent."""
        if self.agent is None:
            raise RuntimeError("No agent loaded.")
        agent = self.agent
        arms = agent.arms
        K = len(arms)
        x = np.asarray(context).reshape(-1)
        dim_x = x.size
        values = np.full((K, B), -np.inf, dtype=float)
        rng_local = np.random.default_rng(int(self.rng.integers(0, 2 ** 31 - 1)))
        for i, a in enumerate(arms):
            try:
                model = agent.models[a]
                Lambda = np.asarray(model.Lambda)
                try:
                    cov = np.linalg.inv(Lambda)
                except Exception:
                    cov = np.linalg.pinv(Lambda)
                mean = _safe_posterior_mean_of_model(model, Lambda.shape[0])
            except Exception:
                mean = np.zeros(dim_x, dtype=float)
                cov = np.eye(dim_x, dtype=float) * 1e-6
            # align shapes
            if cov.shape[0] != dim_x or cov.shape[1] != dim_x:
                if cov.shape[0] >= dim_x and cov.shape[1] >= dim_x:
                    cov = cov[:dim_x, :dim_x]
                else:
                    newcov = np.eye(dim_x, dtype=float) * 1e-6
                    newcov[: cov.shape[0], : cov.shape[1]] = cov
                    cov = newcov
                mean = _safe_align_vec(mean, dim_x)
            else:
                mean = _safe_align_vec(mean, dim_x)
            # belief
            try:
                bel = agent.beliefs[a]
                try:
                    if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(agent, "t", None):
                        bel.compute_posterior(getattr(agent, "t", 0))
                except Exception:
                    pass
                pb_mean = float(getattr(bel, "mean", 0.0))
                pb_std = float(np.sqrt(max(getattr(bel, "var", 1e-12), 1e-12)))
            except Exception:
                pb_mean = 0.0
                pb_std = 1e-6
            try:
                thetas = rng_local.multivariate_normal(mean, cov, size=B)
            except Exception:
                thetas = np.tile(mean.reshape(1, -1), (B, 1))
            preds = thetas @ x
            pb_samples = pb_mean + pb_std * rng_local.standard_normal(size=B)
            pw = float(getattr(agent, "pb_weight", self.pb_weight))
            values[i, :] = preds + pw * pb_samples
        chosen = np.argmax(values, axis=0)
        counts = np.bincount(chosen, minlength=K)
        probas = counts.astype(float) / float(B)
        return {arms[i]: float(probas[i]) for i in range(K)}

    # ---------- Greedy selection (deterministic) ----------
    def greedy_action(self, context: np.ndarray, n_actions: int = 1) -> List[str]:
        """Deterministic greedy selection using posterior means (no sampling)."""
        if self.agent is None:
            raise RuntimeError("No agent loaded.")
        x = np.asarray(context).reshape(-1)
        arms = self.agent.arms
        values = []
        for a in arms:
            try:
                theta = _safe_posterior_mean_of_model(self.agent.models[a], x.size)
            except Exception:
                theta = np.zeros(x.size)
            theta = _safe_align_vec(theta, x.size)
            pred = float(theta @ x)
            pb_mean = 0.0
            try:
                bel = self.agent.beliefs[a]
                try:
                    if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(self.agent, "t", None):
                        bel.compute_posterior(getattr(self.agent, "t", 0))
                except Exception:
                    pass
                pb_mean = float(getattr(bel, "mean", 0.0))
            except Exception:
                pb_mean = 0.0
            val = pred + float(getattr(self.agent, "pb_weight", self.pb_weight)) * pb_mean
            values.append(val)
        # return top-n unique arms (no duplicates)
        idxs = np.argsort(values)[::-1][:n_actions]
        return [arms[i] for i in idxs]

    # ---------- Prediction API ----------
    def predict(self, context: List[float], n_actions: int = 1, stochastic: bool = False) -> List[str]:
        x = np.asarray(context).reshape(-1)
        if stochastic:
            if self.agent is None:
                raise RuntimeError("No agent loaded.")
            return self.agent.act(x, n_actions=n_actions)
        else:
            return self.greedy_action(x, n_actions=n_actions)

    # ---------- Batch updates ----------
    def batch_update(self, updates: List[Dict], log: bool = True):
        """
        updates: list of dicts each with keys:
          - 'a': arm id
          - 'trace': list or array of length w (0/1)
          - 'context': list of length context_dim+1 (including intercept) OR None
          - 't': optional timestep to pass to belief.update (defaults to current agent.t)
          - 'p_b': optional behavior probability to store in log
        """
        if self.agent is None:
            raise RuntimeError("No agent loaded.")

        for u in updates:
            a = u["a"]
            trace = np.atleast_2d(np.asarray(u["trace"], dtype=float))
            context = u.get("context", None)
            if context is None:
                # if user omitted context, use zeros with intercept
                ctx = np.zeros(self.context_dim + 1, dtype=float)
                ctx[0] = 1.0
            else:
                ctx = np.asarray(context, dtype=float).reshape(-1)
            t = int(u.get("t", getattr(self.agent, "t", 0)))
            try:
                self.agent.update(a, trace, ctx)
            except Exception:
                # best-effort update
                try:
                    self.agent.beliefs[a].update(trace, t)
                    self.agent.models[a].update(ctx, float(1.0 + trace.sum()))
                except Exception:
                    pass
            p_b = float(u.get("p_b", 0.0))
            # log it
            entry = SimpleLoggedStep(x=ctx.copy(), a=a, r=float(1.0 + trace.sum()), p_b=p_b, t=t)
            if log:
                self.logged.append(entry)
        return len(updates)

    # ---------- Simulation runner (lightweight) ----------
    def simulate_and_log(self, horizon: int = 180, n_actions: int = 3, propensity_mc: int = 500):
        """
        Runs a simulation using current env + helper + freshly initialized beliefs (or existing agent).
        Returns (regrets_array, entropies_array) and appends to self.logged.
        """
        if self.env is None:
            raise RuntimeError("No environment constructed (need eval pkl).")
        # ensure agent exists; if not create fresh progressive agent
        if self.agent is None:
            if self.helper is None:
                raise RuntimeError("No agent and no helper (priors) available.")
            self.create_fresh_agent(ProgressiveBelief)
        agent = self.agent

        regrets = []
        entropies = []
        rng_local = np.random.default_rng(int(self.rng.integers(0, 2 ** 31 - 1)))
        arms = list(self.env.dists.keys())

        for t in range(horizon):
            # sample random context
            age = float(self.rng.uniform(0, 1))
            device_mobile = int(self.rng.binomial(1, 0.6))
            recent_eng = float(np.tanh(self.rng.exponential(1.0) / 2.0))
            premium = int(self.rng.binomial(1, 0.2))
            context = np.array([1.0, age, device_mobile, recent_eng, premium])

            # pick actions
            actions = agent.act(context, n_actions=n_actions)

            # estimate propensities for logging
            try:
                probas = self.estimate_thompson_probas(context, B=propensity_mc)
            except Exception:
                probas = {a: 1.0 / len(arms) for a in arms}

            # interact
            for a in actions:
                trace = self.env.step(a, t, n_samples=1, context=context)
                r = float(1.0 + trace.sum())
                p_b = max(probas.get(a, 1e-12), 1e-12)
                self.logged.append(SimpleLoggedStep(x=context.copy(), a=a, r=r, p_b=p_b, t=t))
                agent.update(a, trace, context)
            agent.step()

            # diagnostic regret: compare to oracle expected reward under empirical means
            er = np.array([1.0 + self.empirical_means[a].sum() for a in actions])
            r_opt = max(1.0 + v.sum() for v in self.empirical_means.values())
            regrets.append(float(np.mean(r_opt - er)))
            _, counts = np.unique(actions, return_counts=True)
            probs = counts / counts.sum()
            entropies.append(float(-np.sum(probs * np.log(probs + 1e-12))))
        return np.array(regrets), np.array(entropies)

    # ---------- Offline evaluation wrapper ----------
    def compute_ope(self, eval_agent=None, clip=10.0, snips=True, n_boot=200):
        """Compute IPS/SNIPS/DM/DR for the last logged dataset against eval_agent greedy policy."""
        if len(self.logged) == 0:
            raise RuntimeError("No logged data to evaluate on.")
        dataset = self.logged
        arms = list(self.env.dists.keys()) if self.env is not None else list(self.agent.arms)
        dm_model = DirectMethodModel(arms=arms, dim=self.context_dim + 1, alpha=1.0, sigma2=self.sigma2)
        dm_model.fit(dataset)

        if eval_agent is None:
            eval_agent = self.agent

        def eval_action_fn(x):
            return self.greedy_action(np.asarray(x).reshape(-1), n_actions=1)[0]

        metrics = evaluate_offline(dataset=dataset, arms=arms, eval_action_fn=eval_action_fn, dm_model=dm_model, clip=clip, snips=snips, n_boot=n_boot)
        return metrics
