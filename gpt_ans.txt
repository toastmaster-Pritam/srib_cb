"""
api.py - FastAPI endpoints to serve model predictions and run simulations.

Run with:
    uvicorn api:app --reload --host 0.0.0.0 --port 8000
"""

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional, Any
import uvicorn
import numpy as np

from model_service import ModelService

app = FastAPI(title="Contextual Bandit API")
svc = ModelService()  # single instance for admin usage

# -------------------------
# Request / Response models
# -------------------------
class PredictByIdReq(BaseModel):
    context_id: int
    n_actions: Optional[int] = 1
    stochastic: Optional[bool] = False

class PredictReq(BaseModel):
    context: Optional[List[float]] = None
    context_id: Optional[int] = None
    n_actions: Optional[int] = 1
    stochastic: Optional[bool] = False

class SimulateReq(BaseModel):
    context_ids: Optional[List[int]] = None
    horizon: Optional[int] = 60
    n_actions: Optional[int] = 3
    n_trials: Optional[int] = 3
    context_mode: Optional[str] = "random"  # or "round-robin"
    propensity_mc: Optional[int] = 500

class UpdateEntry(BaseModel):
    a: Any
    trace: List[int]
    context: Optional[Any] = None
    t: Optional[int] = None
    p_b: Optional[float] = 0.0
    ctx_id: Optional[int] = None

class UpdateBatchReq(BaseModel):
    updates: List[UpdateEntry]

# -------------------------
# Endpoints
# -------------------------
@app.get("/health")
def health():
    return {"status": "ok"}

@app.get("/contexts")
def get_contexts():
    try:
        return {"contexts": svc.get_context_pool()}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/predict_by_id")
def predict_by_id(req: PredictByIdReq):
    try:
        actions = svc.predict(context_id=req.context_id, n_actions=req.n_actions, stochastic=req.stochastic)
        return {"actions": actions}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@app.post("/predict")
def predict(req: PredictReq):
    try:
        if req.context_id is not None:
            actions = svc.predict(context_id=req.context_id, n_actions=req.n_actions, stochastic=req.stochastic)
        elif req.context is not None:
            actions = svc.predict(context=req.context, n_actions=req.n_actions, stochastic=req.stochastic)
        else:
            actions = svc.predict(n_actions=req.n_actions, stochastic=req.stochastic)
        return {"actions": actions}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@app.post("/simulate_with_contexts")
def simulate(req: SimulateReq):
    try:
        context_ids = req.context_ids
        # run multi-trial simulate
        res = svc.simulate_trials(n_trials=req.n_trials, horizon=req.horizon, n_actions=req.n_actions, propensity_mc=req.propensity_mc, selected_context_ids=context_ids, context_mode=req.context_mode)
        # convert numpy arrays to lists for json
        return {
            "cumavg_mean": res["cumavg_mean"].tolist(),
            "cumavg_std": res["cumavg_std"].tolist(),
            "regrets_shape": list(res["regrets"].shape),
        }
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@app.post("/update_batch")
def update_batch(req: UpdateBatchReq):
    try:
        updates = [u.dict() for u in req.updates]
        n = svc.batch_update(updates, log=True)
        return {"updated": n}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@app.get("/ope")
def ope(n_boot: int = 200):
    try:
        metrics = svc.compute_ope(n_boot=n_boot)
        # convert numpy objects to floats/lists where needed
        for k, v in list(metrics.items()):
            if isinstance(v, dict):
                for kk in ["value", "boot_mean"]:
                    if kk in v:
                        v[kk] = float(v[kk])
                if "ci95" in v:
                    v["ci95"] = [float(x) for x in v["ci95"]]
        return metrics
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

# Quick dev-run
if __name__ == "__main__":
    uvicorn.run("api:app", host="0.0.0.0", port=8000, reload=True)
