"""
app.py - Streamlit Admin UI

Run:
    streamlit run app.py

This UI uses ModelService in-process (no separate API by default).
Features:
 - context pool selection & editing (simple)
 - predict (greedy / stochastic) for selected contexts
 - simulate with selected contexts, plot regret curve (mean ± std over trials)
 - compute OPE on the logged dataset
"""

import streamlit as st
import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
import json
from typing import List

from model_service import ModelService, context_to_vector, context_label

st.set_page_config(layout="wide", page_title="Contextual Bandit Admin")

# ensure session service
if "svc" not in st.session_state:
    st.session_state["svc"] = ModelService()

svc: ModelService = st.session_state["svc"]

st.title("Contextual Bandit — Admin Console")

left, right = st.columns([2, 3])

with left:
    st.header("Context pool")
    pool = svc.get_context_pool()
    df_pool = pd.DataFrame(pool)
    # show as selectable list (multi-select uses labels)
    options = [f"{row['id']}: {row['label']}" for _, row in df_pool.iterrows()]
    selected = st.multiselect("Select contexts (one or more)", options, default=options[:3])

    # parse selected ids
    selected_ids = []
    for s in selected:
        try:
            idx = int(s.split(":")[0])
            selected_ids.append(idx)
        except Exception:
            pass

    st.markdown("**Edit / Add context**")
    with st.form("ctx_form"):
        age = st.slider("age", 0.0, 1.0, 0.5)
        device_mobile = st.selectbox("device_mobile", options=[0, 1], index=1)
        premium = st.selectbox("premium", options=[0, 1], index=0)
        recent_eng = st.slider("recent_eng", 0.0, 1.0, 0.2)
        add_ctx = st.form_submit_button("Add context to pool")
    if add_ctx:
        ctx = {"age": float(age), "device_mobile": int(device_mobile), "recent_eng": float(recent_eng), "premium": int(premium)}
        new_id = svc.add_context_to_pool(ctx)
        st.success(f"Added context id={new_id}")
        st.experimental_rerun()

    st.markdown("---")
    st.header("Agent")
    saved_agents = [f for f in st.sidebar.empty() or []]  # placeholder to avoid linter
    # list pickles in data/
    import os
    saved_agents = [f for f in os.listdir("data") if f.endswith(".pkl")]
    sel_agent = st.selectbox("Load agent (pickle)", ["-- none --"] + saved_agents)
    if st.button("Load selected agent") and sel_agent != "-- none --":
        try:
            svc.load_agent(os.path.join("data", sel_agent))
            st.experimental_rerun()
        except Exception as e:
            st.error(f"Failed to load agent: {e}")

    if st.button("Create fresh Progressive agent"):
        try:
            svc.create_fresh_agent()
            st.success("Created fresh Progressive agent.")
        except Exception as e:
            st.error(str(e))

    if st.button("Clear logged data"):
        svc.logged = []
        st.success("Cleared logged data")

with right:
    st.header("Predict")
    n_actions = st.number_input("n_actions (per context)", min_value=1, max_value=20, value=3)
    stochastic = st.checkbox("Stochastic (Thompson)", value=False)

    if st.button("Predict for selected contexts"):
        if len(selected_ids) == 0:
            st.warning("Select at least one context")
        else:
            rows = []
            for idx in selected_ids:
                try:
                    ctx = svc.context_pool[idx]
                    vec = context_to_vector(ctx)
                    try:
                        acts = svc.predict(context_id=idx, n_actions=n_actions, stochastic=stochastic)
                    except Exception as e:
                        acts = f"err: {e}"
                    rows.append({"context_id": idx, "label": context_label(ctx), "actions": acts})
                except Exception as e:
                    rows.append({"context_id": idx, "label": "ERROR", "actions": str(e)})
            st.table(pd.DataFrame(rows))

    st.markdown("---")
    st.header("Simulation & Regret")
    sim_horizon = st.number_input("horizon (rounds)", min_value=10, max_value=1000, value=60)
    sim_trials = st.number_input("n_trials", min_value=1, max_value=20, value=3)
    sim_mode = st.selectbox("context sampling", options=["random", "round-robin"], index=0)
    run_sim = st.button("Run simulation & plot regret")

    if run_sim:
        if len(selected_ids) == 0:
            st.warning("Select contexts to simulate with (or add some).")
        else:
            with st.spinner("Running simulations..."):
                res = svc.simulate_trials(n_trials=int(sim_trials), horizon=int(sim_horizon), n_actions=int(n_actions), propensity_mc=300, selected_context_ids=selected_ids, context_mode=sim_mode)
            mean_curve = res["cumavg_mean"]
            std_curve = res["cumavg_std"]
            xs = np.arange(1, len(mean_curve) + 1)
            fig, ax = plt.subplots(figsize=(9, 4))
            ax.plot(xs, mean_curve, label="mean cum. avg. regret")
            ax.fill_between(xs, mean_curve - std_curve, mean_curve + std_curve, alpha=0.12)
            ax.set_xlabel("Round")
            ax.set_ylabel("Average cumulative regret")
            ax.legend()
            ax.grid(alpha=0.3)
            st.pyplot(fig)
            st.success("Simulation complete; logged interactions appended to service logged buffer.")

    st.markdown("---")
    st.header("OPE on logged data")
    if st.button("Compute OPE (DR/IPS/DM)"):
        if len(svc.logged) == 0:
            st.warning("No logged data available. Run simulation first or upload logs.")
        else:
            with st.spinner("Computing OPE (bootstrap)..."):
                try:
                    metrics = svc.compute_ope(n_boot=200)
                    st.json(metrics)
                except Exception as e:
                    st.error(str(e))

    st.markdown("---")
    st.header("Recent logged interactions")
    if len(svc.logged) > 0:
        rows = []
        for e in svc.logged[-50:]:
            rows.append({"a": e.a, "r": e.r, "p_b": e.p_b, "t": e.t, "ctx_id": e.ctx_id})
        st.table(pd.DataFrame(rows))

st.markdown("---")
st.write("Notes: Use the left panel to manage contexts and the agent. Use the right panel for prediction, simulation and OPE.")
