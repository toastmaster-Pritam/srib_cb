class DummyAgent:
    """A simple random (non-learning) agent for baseline comparisons."""
    def __init__(self, arms, seed: int = 0):
        self.arms = list(arms)
        self._rng = np.random.default_rng(int(seed))

    def act(self, context, n_actions=1):
        # sample without replacement if n_actions <= K
        k = len(self.arms)
        n = int(n_actions)
        if n >= k:
            # if asking more or equal than arms, return all in a random order
            return list(self._rng.choice(self.arms, size=k, replace=False))
        return list(self._rng.choice(self.arms, size=n, replace=False))

    # Optional: Minimal placeholder methods so code checking hasattr works cleanly
    def update(self, *args, **kwargs):
        # dummy agent does not update
        return

    def step(self):
        return

def generate_trace_for_arm_using_empirical(arm, context, n_samples=1, trial_seed: int = None):
    """
    Draw n_samples traces for this arm from its empirical distribution in env.dists.
    If trial_seed provided, use deterministic RNG for light augmentation so results are reproducible.
    """
    # base draws (EmpiricalDistribution has its own RNG seeded at construction)
    draws = env.dists[arm].sample(n=n_samples)

    # context modulated scaling & optional deterministic flips
    try:
        ctx = np.asarray(context).reshape(-1)
    except Exception:
        ctx = np.zeros(CONTEXT_DIM + 1)
    mu = float(true_theta[arm] @ ctx)
    scale = 0.5 * (1.0 / (1.0 + np.exp(-mu))) + 0.5

    # deterministic flipping based on trial seed (so everything is reproducible).
    flip_p = max(0.0, (scale - 0.5) * 0.2)
    if flip_p > 0 and trial_seed is not None:
        # deterministic RNG per trial + arm
        arm_hash = abs(hash(str(arm))) % 1000000
        rng_local = np.random.default_rng(int(trial_seed + arm_hash))
        flips = rng_local.random(draws.shape) < flip_p
        draws = np.clip(draws + flips.astype(float), 0.0, 1.0)

    return draws.astype(float)

def estimate_thompson_probas(agent, context, B=PROP_MC, rng_local=None, agent_seed: int = None):
    """
    Estimate Thompson propensities. If rng_local is None, create one using agent_seed for determinism.
    """
    if rng_local is None:
        if agent_seed is None:
            # fallback to a default generator (but prefer agent_seed)
            rng_local = np.random.default_rng(123456)
        else:
            rng_local = np.random.default_rng(int(agent_seed + 123456))
    arms = agent.arms
    K = len(arms)
    x = np.asarray(context).reshape(-1)
    dim_x = x.size
    values = np.full((K, B), -np.inf, dtype=float)

    for i, a in enumerate(arms):
        try:
            model = agent.models[a]
            Lambda = np.asarray(model.Lambda)
            try:
                cov = np.linalg.inv(Lambda)
            except Exception:
                cov = np.linalg.pinv(Lambda + np.eye(Lambda.shape[0]) * 1e-9)
            mean = _safe_align(model.posterior_mean(), dim_x)
        except Exception:
            mean = np.zeros(dim_x, dtype=float)
            cov = np.eye(dim_x, dtype=float) * 1e-6

        try:
            bel = agent.beliefs[a]
            try:
                if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(agent, "t", None):
                    bel.compute_posterior(getattr(agent, "t", 0))
            except Exception:
                pass
            pb_mean = float(getattr(bel, "mean", 0.0))
            pb_var = float(getattr(bel, "var", 0.0))
            pb_std = math.sqrt(max(pb_var, 1e-12))
        except Exception:
            pb_mean = 0.0
            pb_std = 1e-6

        try:
            thetas = rng_local.multivariate_normal(mean, cov, size=B)
            thetas = np.atleast_2d(thetas)
            if thetas.shape[1] != dim_x:
                tmp = np.zeros((B, dim_x))
                tmp[:, : thetas.shape[1]] = thetas[:, :min(thetas.shape[1], dim_x)]
                thetas = tmp
        except Exception:
            thetas = np.tile(mean.reshape(1, -1), (B, 1))

        preds = thetas @ x
        pb_samples = pb_mean + pb_std * rng_local.standard_normal(size=B)
        pw = float(getattr(agent, "pb_weight", 1.0))
        values[i, :] = preds + pw * pb_samples

    chosen = np.argmax(values, axis=0)
    counts = np.bincount(chosen, minlength=K)
    probas = counts.astype(float) / float(B)
    return {arms[i]: float(probas[i]) for i in range(K)}

def bandit_trial_with_logging(env, beliefs, horizon, n_actions, policy_type="Progressive", agent_seed: int = 0, pb_weight: float = 0.6, generator_fn=None, propensity_mc=PROP_MC):
    """
    Train/log under a selected policy ('Progressive', 'ContextOnly', 'Dummy').
    Deterministic seeds used: agent_seed controls RNGs inside.
    """
    # Build agent depending on policy
    if policy_type == "Dummy":
        # DummyAgent picks random arms; no learning
        agent = DummyAgent(env.dists.keys(), seed=agent_seed)
    else:
        # Contextual Bayesian agent (Progressive or ContextOnly)
        agent = ContextualBayesianBandit(beliefs, dim=CONTEXT_DIM + 1, alpha=1.0, sigma2=0.25, pb_weight=pb_weight, seed=int(agent_seed))

    regrets = []
    entropies = []
    logged_dataset = []

    # deterministic RNG for trial-level randomness (used for augmentation & MC where needed)
    trial_rng = np.random.default_rng(int(agent_seed + 77777))

    for t in range(horizon):
        context = sample_context()
        actions = agent.act(context, n_actions=n_actions)

        # Estimate propensities deterministically
        if policy_type == "Dummy":
            probas = {a: 1.0 / len(agent.arms) for a in agent.arms}
        else:
            probas = estimate_thompson_probas(agent, context, B=propensity_mc, rng_local=np.random.default_rng(int(agent_seed + t + 4242)), agent_seed=agent_seed)

        # interact and log: for each selected action, sample full empirical trace (generator_fn) and update
        for a in actions:
            p_b = max(probas.get(a, 0.0), 1e-12)
            # pass deterministic trial_seed to generator so augmentation is reproducible
            trace = generator_fn(a, context, n_samples=1, trial_seed=int(agent_seed + t))
            r = float(1.0 + trace.sum())
            logged_dataset.append(LoggedStep(x=context.copy(), a=a, r=r, p_b=p_b, t=t))

            # update agent only if it supports learning (skip Dummy)
            if hasattr(agent, "update") and policy_type != "Dummy":
                agent.update(a, trace, context)

        # step for learning agents only
        if hasattr(agent, "step") and policy_type != "Dummy":
            agent.step()

        # compute regret w.r.t. oracle expected reward (approx using empirical mean per item)
        true_rewards = np.array([1.0 + item_traces[a].mean(axis=0).sum() if a in item_traces else 0.0 for a in actions])
        r_opt = max((1.0 + item_traces[it].mean(axis=0).sum()) for it in item_traces.keys())
        regrets.append(np.mean(r_opt - true_rewards))

        _, counts = np.unique(actions, return_counts=True)
        probs = counts / counts.sum()
        entropies.append(-np.sum(probs * np.log(probs + 1e-12)))

    return np.array(regrets), np.array(entropies), agent, logged_dataset

def make_eval_fn(agent):
    # deterministic eval RNG if needed
    eval_rng = np.random.default_rng(SEED + 9999)
    def eval_action_fn(x):
        x = np.asarray(x).reshape(-1)
        # If agent lacks models (Dummy), return a random arm (deterministic RNG)
        if not hasattr(agent, "models"):
            return eval_rng.choice(list(arms))
        best_val = -np.inf
        best_arm = None
        for a in arms:
            try:
                theta = agent.models[a].posterior_mean()
                theta = np.asarray(theta).reshape(-1)
                theta = _safe_align(theta, x.size)
            except Exception:
                theta = np.zeros(x.size)
            pred = float(theta @ x)
            pb_mean = 0.0
            try:
                bel = agent.beliefs[a]
                try:
                    if getattr(bel, "has_new_obs", False) or getattr(bel, "t_post", None) != getattr(agent, "t", None):
                        bel.compute_posterior(getattr(agent, "t", 0))
                except Exception:
                    pass
                pb_mean = float(getattr(bel, "mean", 0.0))
            except Exception:
                pb_mean = 0.0
            val = pred + float(getattr(agent, "pb_weight", 0.0)) * pb_mean
            if val > best_val:
                best_val = val
                best_arm = a
        return best_arm
    return eval_action_fn
