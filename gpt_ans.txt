# compute regret diagnostic (use empirical_means)
            try:
                true_rewards = np.array([1.0 + self.empirical_means[a].sum() for a in actions])
                r_opt = max(1.0 + v.sum() for v in self.empirical_means.values())
            except Exception:
                # fallback: treat 0
                true_rewards = np.zeros(len(actions), dtype=float)
                r_opt = 0.0
            regrets.append(float(np.mean(r_opt - true_rewards)))
            _, counts = np.unique(actions, return_counts=True)
            probs = counts / counts.sum()
            entropies.append(float(-np.sum(probs * np.log(probs + 1e-12))))

        # append logged_run to global log
        self.logged.extend(logged_run)
        return np.array(regrets), np.array(entropies), logged_run

    # ---------------------
    # Multi-trial simulate to compute mean/std regret curves
    # ---------------------
    def simulate_trials(self, n_trials: int = 3, horizon: int = 180, n_actions: int = 3, propensity_mc: int = 500, selected_context_ids: Optional[List[int]] = None, context_mode: str = "random") -> Dict[str, Any]:
        """
        Run multiple independent trials (fresh agent for each) and return aggregated regret statistics.
        Returns dict with keys: 'regrets' (n_trials x horizon), 'mean_cum_regret' (horizon,), 'std_cum_regret' (horizon,)
        """
        all_regrets = []
        for i in range(n_trials):
            # clear agent and use fresh agent to ensure independent trials
            self.agent = None
            reg, ent, logged_run = self.simulate_and_log(horizon=horizon, n_actions=n_actions, propensity_mc=propensity_mc, selected_context_ids=selected_context_ids, context_mode=context_mode, fresh_agent=True)
            all_regrets.append(reg)
        arr = np.vstack(all_regrets)  # (n_trials, horizon)
        # compute cumulative average regret per trial, then mean/std across trials
        xs = np.arange(1, arr.shape[1] + 1)
        cumavg = np.cumsum(arr, axis=1) / xs
        mean = np.mean(cumavg, axis=0)
        std = np.std(cumavg, axis=0)
        return {"regrets": arr, "cumavg_mean": mean, "cumavg_std": std}

    # ---------------------
    # OPE wrapper
    # ---------------------
    def compute_ope(self, eval_agent=None, clip=10.0, snips=True, n_boot: int = 200):
        if len(self.logged) == 0:
            raise RuntimeError("No logged data available")
        dataset = self.logged
        arms = list(self.env.dists.keys()) if self.env is not None else list(self.agent.arms)
        dm_model = DirectMethodModel(arms=arms, dim=self.context_dim + 1, alpha=1.0, sigma2=self.sigma2)
        dm_model.fit(dataset)
        if eval_agent is None:
            eval_agent = self.agent

        def eval_action_fn(x):
            x = np.asarray(x).reshape(-1)
            # use greedy selection
            return self.greedy_action(x, n_actions=1)[0]

        metrics = evaluate_offline(dataset=dataset, arms=arms, eval_action_fn=eval_action_fn, dm_model=dm_model, clip=clip, snips=snips, n_boot=n_boot)
        return metrics
